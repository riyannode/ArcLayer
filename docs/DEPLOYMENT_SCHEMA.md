### 2. `agent_bridge_events` — bridge events

```sql
CREATE TABLE IF NOT EXISTS agent_bridge_events (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  session_id text NOT NULL,
  role text NOT NULL,
  agent_id text NOT NULL,
  event_type text NOT NULL,
  event_dedupe_key text,
  payload jsonb NOT NULL,
  payload_hash text NOT NULL,
  job_id text,
  category text,
  runtime_id text,
  created_at timestamptz DEFAULT now()
);

-- Partial unique index for content-event dedup
CREATE UNIQUE INDEX IF NOT EXISTS uq_agent_bridge_events_dedup
  ON agent_bridge_events (event_dedupe_key)
  WHERE event_dedupe_key IS NOT NULL;
```

### 3. `agent_bridge_receipts` — bridge payment receipts

```sql
CREATE TABLE IF NOT EXISTS agent_bridge_receipts (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  session_id text NOT NULL,
  event_id bigint REFERENCES agent_bridge_events(id),
  receipt_type text NOT NULL,
  payload jsonb NOT NULL,
  payment_id text,
  tx_hash text,
  created_at timestamptz DEFAULT now()
);
```

### 4. `external_agent_runtimes` — external PM2 agent registration

```sql
CREATE TABLE IF NOT EXISTS external_agent_runtimes (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  runtime_id text NOT NULL UNIQUE,
  agent_id text NOT NULL,
  label text,
  metadata jsonb DEFAULT '{}',
  last_heartbeat_at timestamptz DEFAULT now(),
  created_at timestamptz DEFAULT now()
);
```

### 5. `x402_resource_payments` / `x402_native_payments` / `x402_gateway_payments`

```sql
CREATE TABLE IF NOT EXISTS x402_resource_payments (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  payment_id text NOT NULL UNIQUE,
  resource text NOT NULL,
  payer text NOT NULL,
  pay_to text NOT NULL,
  amount text NOT NULL,
  status text NOT NULL DEFAULT 'pending',
  session_id text,
  scope text,
  role text,
  chain_id text,
  asset text,
  consumed_at timestamptz,
  settled_at timestamptz,
  created_at timestamptz DEFAULT now()
);

CREATE TABLE IF NOT EXISTS x402_native_payments (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  payment_id text NOT NULL UNIQUE,
  payer text NOT NULL,
  pay_to text NOT NULL,
  amount text NOT NULL,
  status text NOT NULL DEFAULT 'pending',
  chain_id text,
  asset text,
  consumed_at timestamptz,
  settled_at timestamptz,
  created_at timestamptz DEFAULT now()
);

CREATE TABLE IF NOT EXISTS x402_gateway_payments (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  payment_id text NOT NULL UNIQUE,
  payer text NOT NULL,
  status text NOT NULL DEFAULT 'pending',
  consumed_at timestamptz,
  settled_at timestamptz,
  created_at timestamptz DEFAULT now()
);
```

> Note: `x402_native_claim_payment`, `x402_native_consume_payment`, `x402_gateway_claim_settlement`, and `x402_gateway_consume_payment` are **SQL functions** (not tables), defined in the migration files. They are called via `supabase.rpc()`. Do not create them as tables.
